(EP.ComputeWinnersAndLosers
  [LAMBDA (self)                                                (* rao "23-Apr-86 19:05")
    (for LeftSlot in (QUOTE (SpadesLeft HeartsLeft DiamondsLeft ClubsLeft)) as WinnerSlot
       in (QUOTE (SpadeWinner HeartWinner DiamondWinner ClubWinner)) as LoserSlot

(EP.DoPass
  [LAMBDA (Self PCards)                                         (* hed "30-Apr-86 20:59")
    (LET* ((CurrentPassCards (GET.VALUE Self (QUOTE PassCards)))
           [NewNPC (PLUS (FLENGTH PCards))

           (PCards (for PC in PCards collect PC unless (CardList.Member PC CurrentPassCards)))

(EP.Equivalent?
  [LAMBDA (self Card1 Card2 Not?)                               (* HED "25-Apr-86 00:48")
    (if (EQUAL (Card.Suit Card1)
               (Card.Suit Card2))
        then [LET* ((suit (Card.Suit Card1))
                    (PlayedCards (CardList.CardsOfSuit (GET.VALUES self (QUOTE CardsPlayed))
                                                       (Card.Suit Card1)))
                    (HandCards (Hand.CardsInSuit (GET.VALUE (GET.VALUE self (QUOTE Player))
                                                            (QUOTE Hand))
                                                 suit)))
               (if (Card.Higher? Card1 Card2)
                   then (swap Card1 Card2))
               (if Not?
                   then [for val in (Card.ValueSublist (Card.Value Card1)
                                                       (Card.Value Card2))
                           thereis (LET ((testcard (Card.Create val suit)))
                                     (NOT (OR (CardList.Member testcard PlayedCards)
                                              (CardList.Member testcard HandCards]
                 else (for val in (Card.ValueSublist (Card.Value Card1)
                                                     (Card.Value Card2))
                         always (LET ((testcard (Card.Create val suit)))
                                  (OR (CardList.Member testcard PlayedCards)
                                      (CardList.Member testcard HandCards]
      else (SHOULDNT (CONCAT "EP.Equivalent: Cards " Card1 " and " Card2 " not of same suit."])
